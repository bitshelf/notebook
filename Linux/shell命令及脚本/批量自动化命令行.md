# Android
1. 批量向节点写入 $1$ #type
	~~~shell
	for f in *io*; do echo 1 > $f;done
	for f in *io*;do echo -n "$f ";cat $f;done
	~~~

2. 查看 gpio 节点名节点值 #type 
~~~shell
for node in gpio*;
do 
    nodename=`echo $node`
    value=`cat $node`
done
~~~

3. adb push 批量文件 #type 
~~~shell
adb push . /data/
adb push dir\. /data
~~~
