# 自旋锁
~~~c
spinlock_t lock;
spin_lock_init(&lock);
spin_lock(&lock); //获取自旋锁，保护临界区
//临界区
spin__unlock(&lock);//解锁
~~~
* 忙等待，无调度开销
* 进程抢占被禁止
* 锁定期间不能睡眠
# 信号量
~~~c
DECLARE_MUTE(mount_sem); //定义信号量
down(&mount_sem);//获取信号量，保护临界区
critical_section(); //临界区
up(&mount_sem);//释放信号量
~~~
* 拿不到就切换进程，有调度开销
* 锁定期间可以睡眠，不可用于中断上下文
