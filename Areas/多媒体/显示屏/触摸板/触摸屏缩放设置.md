# 最大打印点与屏幕分辨率不对应，缩放做大最大分辨率
`/linux-3.10/drivers/input/touchscreen/gt9xxnew/gt9xx.c`
~~~c
 static void gtp_touch_down (struct goodix_ts_data* ts,s32 id,s32 x,s32 y,s32 w)

 {

 dprintk(DEBUG_X_Y_INFO, "source data:ID:%d, X:%d, Y:%d, W:%d\n", id, x, y, w);

  

 //rpdzkj add for scale max resolution

 if ((SCREEN_MAX_X != ts->abs_x_max) || (SCREEN_MAX_Y != ts->abs_y_max)) {

 GTP_DEBUG("max resolution: %dx%d and registered resolution: %dx%d is diff, try to scale...",

 SCREEN_MAX_X, SCREEN_MAX_Y, ts->abs_x_max, ts->abs_y_max);

 x = x * SCREEN_MAX_X / ts->abs_x_max;

 y = y * SCREEN_MAX_Y / ts->abs_y_max;

 }
 ~~~
